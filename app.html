DIE KERN-RECHENLOGIK (Hier lebt deine VBA-Logik) function berechneImmobilie() { // Eingaben sicher auslesen (verhindert NaN Fehler) const kaufpreis = parseFloat(document.getElementById('kaufpreis').value) || 0; const eigenkapital = parseFloat(document.getElementById('eigenkapital').value) || 0; // Platzhalter für Kaufnebenkosten (z.B. 8%) - später aus Admin-DB holen! const nk_quote = 0.08; const kaufpreis_inkl_nk = kaufpreis * (1 + nk_quote); // Kapitalbedarf berechnen let fremdkapitalbedarf = kaufpreis_inkl_nk - eigenkapital; if(fremdkapitalbedarf < 0) fremdkapitalbedarf = 0; // Darlehen 1 (Hauptdarlehen) Kredithöhe automatisch ins Feld eintragen // (Später ziehst du hier noch KFW Kredite ab) document.getElementById('d1Kredit').value = Math.round(fremdkapitalbedarf); // Platzhalter-Rechnung für Cashflow (Hier baust du deine Zins/AfA/Steuer Logik ein) const zinsSatz = parseFloat(document.getElementById('d1Zins').value) || 3.9; const tilgSatz = parseFloat(document.getElementById('d1Tilgung').value) || 1.5; const jahresAufwand = fremdkapitalbedarf * ((zinsSatz + tilgSatz) / 100); const fiktiveMiete = kaufpreis * 0.04; // 4% Mietrendite Annahme const cashflowJ1 = fiktiveMiete - jahresAufwand; // Simpel: Miete - Rate return { kaufpreisInklNk: kaufpreis_inkl_nk, kapitalbedarf: fremdkapitalbedarf, cashflowJahr1: cashflowJ1 }; } // 5. '#d9534f' : '#28a745'; } // 6. EVENT LISTENER: Bei jedem Tastenanschlag rechnen document.querySelectorAll('#calcForm input').forEach(el => { el.addEventListener('input', liveBerechnung); }); // 7.
